### Especificación de Implementación Frontend (Vite/React)

#### 1. Nuevo hook `src/hooks/useKinMaya.ts`
```typescript
import { useState, useEffect } from 'react';

interface KinMaya {
  fecha: string;
  kin: number;
  tono: number;
  nombre_tono: string;
  palabraclave_tono: string;
  sello: number;
  nombre_sello: string;
  palabraclave_sello: string;
  color: string;
}

export function useKinMaya(fecha?: string) {
  const [kin, setKin] = useState<KinMaya | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fechaParam = fecha ?? new Date().toISOString().split('T')[0];
    fetch(`https://api.romensuarez.com/maya/kin?fecha=${fechaParam}`)
      .then(r => r.json())
      .then(data => { setKin(data); setLoading(false); })
      .catch(() => setLoading(false));
  }, [fecha]);

  return { kin, loading };
}
```

#### 2. Nuevo componente `src/components/KinMayaWidget.tsx`
```tsx
import React from 'react';
import { useKinMaya } from '../hooks/useKinMaya';

const COLOR_MAP: Record<string, string> = {
  'Amarillo': 'bg-amber-50 border-amber-200 text-amber-800',
  'Rojo':     'bg-rose-50  border-rose-200  text-rose-800',
  'Blanco':   'bg-stone-50 border-stone-200 text-stone-700',
  'Azul':     'bg-sky-50   border-sky-200   text-sky-800',
};

export function KinMayaWidget() {
  const { kin, loading } = useKinMaya();

  if (loading) return (
    <div className="h-16 bg-stone-50 rounded-2xl animate-pulse border border-[#EAE2D6]" />
  );
  if (!kin) return null;

  const colorClass = COLOR_MAP[kin.color] ?? COLOR_MAP['Blanco'];

  return (
    <div className={`flex items-center gap-4 px-5 py-3 rounded-2xl border ${colorClass}`}>
      <div className="text-center min-w-[48px]">
        <div className="text-2xl font-bold leading-none">{kin.kin}</div>
        <div className="text-xs opacity-60 mt-0.5">Kin</div>
      </div>
      <div className="w-px h-8 bg-current opacity-20" />
      <div className="flex-1 min-w-0">
        <div className="font-semibold text-sm leading-tight truncate">
          {kin.nombre_sello} · {kin.nombre_tono}
        </div>
        <div className="text-xs opacity-60 mt-0.5">{kin.palabraclave_tono} · {kin.color}</div>
      </div>
      <div className="text-xs opacity-40 font-medium">🌀 Hoy</div>
    </div>
  );
}
```
